def sha256(data):
    return hashlib.sha256(data).hexdigest()[:16]

def log(f, msg):
    f.write(msg + "\n")
    f.flush()

def try_read(path):
    try:
        data = open(path, "rb").read()
        return data, None
    except Exception as e:
        return None, str(e)

def try_write(path, content):
    try:
        with open(path, "wb") as wf:
            wf.write(content)
        return True, None
    except Exception as e:
        return False, str(e)

def try_append(path, content):
    try:
        with open(path, "ab") as wf:
            wf.write(content)
        return True, None
    except Exception as e:
        return False, str(e)

def try_cmd(cmd, timeout=5):
    try:
        r = subprocess.run(cmd, shell=True, capture_output=True, text=True, timeout=timeout)
        return r.stdout.strip(), r.stderr.strip(), r.returncode
    except Exception as e:
        return "", str(e), -1
